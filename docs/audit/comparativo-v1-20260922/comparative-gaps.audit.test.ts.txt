function request(payload: unknown = body): ApiRequest {
  return { method: 'POST', headers: { origin, 'content-type': 'application/json', 'x-vercel-forwarded-for': '203.0.113.42' }, body: payload };
}
function response() {
  const state = { status: 0, body: undefined as unknown };
  const value: ApiResponse = { setHeader() {}, status(code) { state.status = code; return value; }, json(data) { state.body = data; } };
  return { state, value };
}
function harness() {
  let rows: unknown[] = [{ id, slug: 'kit-auditoria', name: 'Kit', sku: 'AUD-1', min_quantity: 5, primary_image_url: 'https://example.invalid/kit.webp' }];
  let storedHash: string | undefined;
  let rpcCalls = 0;
  const fetcher = vi.fn(async (url: string | URL, init?: RequestInit) => {
    if (String(url).includes('/v_site_products_public')) return Response.json(rows);
    if (!String(url).endsWith('/create_site_quote_request')) throw new Error('Unexpected audit endpoint');
    rpcCalls++;
    const hash = JSON.parse(String(init?.body)).p_request_meta.requestHash as string;
    if (storedHash && storedHash !== hash) return Response.json({ message: 'client_request_id_conflict' }, { status: 409 });
    const duplicate = Boolean(storedHash);
    storedHash = hash;
    return Response.json({ requestId: id, duplicate });
  });
  return { fetcher, withdraw: () => { rows = []; }, changeImage: () => { rows = (rows as Record<string, unknown>[]).map(r => ({ ...r, primary_image_url: 'https://example.invalid/updated.webp' })); }, calls: () => rpcCalls };
}
describe('Comparative audit: independently reproduced gaps', () => {
  beforeEach(() => {
    vi.useFakeTimers(); vi.setSystemTime(new Date('2026-09-22T12:00:00Z'));
    vi.stubEnv('SITE_SUPABASE_URL', 'https://xlzmclcjdncjfdrjxclt.supabase.co');
    vi.stubEnv('SITE_SUPABASE_SECRET_KEY', `sb_secret_${'x'.repeat(40)}`);
    vi.stubEnv('SITE_REQUEST_HASH_SALT', 'synthetic-salt-more-than-thirty-two-characters');
    vi.stubEnv('SITE_PUBLIC_ORIGIN', origin);
    vi.stubEnv('CATALOG_SUPABASE_PUBLISHABLE_KEY', 'sb_publishable_synthetic');
  });
  afterEach(() => { vi.useRealTimers(); vi.unstubAllEnvs(); vi.unstubAllGlobals(); });
  it('GAP-01: saved request becomes 422 after catalog withdrawal', async () => {
    const h = harness(); vi.stubGlobal('fetch', h.fetcher);
    const first = response(); await quoteHandler(request(), first.value); expect(first.state.status).toBe(201);
    h.withdraw(); const retry = response(); await quoteHandler(request(), retry.value);
    expect(retry.state.status).toBe(422); expect(h.calls()).toBe(1);
  });
  it('GAP-02: unchanged customer intent conflicts after editorial image change', async () => {
    const h = harness(); vi.stubGlobal('fetch', h.fetcher);
    const first = response(); await quoteHandler(request(), first.value); expect(first.state.status).toBe(201);
    h.changeImage(); const retry = response(); await quoteHandler(request(), retry.value);
    expect(retry.state.status).toBe(409); expect(h.calls()).toBe(2);
  });
  it('CONTROL: unchanged intent and catalog recover duplicate successfully', async () => {
    const h = harness(); vi.stubGlobal('fetch', h.fetcher);
    const first = response(); await quoteHandler(request(), first.value);
    const retry = response(); await quoteHandler(request(), retry.value);
    expect([first.state.status, retry.state.status]).toEqual([201, 200]);
  });
  it('GAP-03: proposal signing uses site_api credential despite separate Storage secret', async () => {
    const jwt = `${Buffer.from(JSON.stringify({ alg: 'HS256', typ: 'JWT' })).toString('base64url')}.${Buffer.from(JSON.stringify({ role: 'site_api' })).toString('base64url')}.synthetic_signature`;
    vi.stubEnv('SITE_SUPABASE_SERVICE_JWT', jwt);
    let signingCredential = '';
    vi.stubGlobal('fetch', vi.fn(async (url: string | URL, init?: RequestInit) => {
      if (String(url).includes('/rpc/')) return Response.json({ bucket: 'customer-proposals', path: 'audit/proposal.pdf' });
      signingCredential = (init?.headers as Record<string, string>).Authorization;
      return Response.json({ signedURL: '/object/sign/customer-proposals/audit/proposal.pdf?token=synthetic' });
    }));
    const req = request({ proposalId: id }); req.headers.authorization = `Bearer ${'a'.repeat(40)}`;
    const res = response(); await proposalHandler(req, res.value);
    expect(res.state.status).toBe(200); expect(signingCredential).toBe(`Bearer ${jwt}`);
  });
});
